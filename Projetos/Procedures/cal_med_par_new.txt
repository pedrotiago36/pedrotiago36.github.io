IF OBJECT_ID('dbo.CAL_MED_PAR', 'P') IS NOT NULL
    DROP PROCEDURE dbo.CAL_MED_PAR
GO

CREATE PROCEDURE dbo.CAL_MED_PAR
    @V_Is_Mat_Ini INT,
    @V_Is_Mat_Fin INT,
    @V_Is_Tur     SMALLINT,
    @V_Nr_Bim     SMALLINT,
    @V_Is_Usu     SMALLINT
AS
-- exec dbo.CAL_MED_PAR 65322, 65322, 4378, 4, 34
SET NOCOUNT ON;
SET XACT_ABORT ON;

BEGIN TRY
  BEGIN TRANSACTION;

  /* ---[ Variáveis locais — evita parameter sniffing ]--------------- */
  DECLARE
    @Is_Mat_Ini INT,
    @Is_Mat_Fin INT,
    @Is_Tur     SMALLINT,
    @Nr_Bim     SMALLINT,
    @Is_Usu     SMALLINT,
    @Is_Anl     SMALLINT,
    @SerIni     SMALLINT,
    @Cd_Ser     SMALLINT;

  SELECT
    @Is_Mat_Ini = @V_Is_Mat_Ini,
    @Is_Mat_Fin = @V_Is_Mat_Fin,
    @Is_Tur     = @V_Is_Tur,
    @Nr_Bim     = @V_Nr_Bim,
    @Is_Usu     = @V_Is_Usu;

  /* ---[ Ano letivo, série e faixa de série inicial — 1 única query ] */
  SELECT
    @Is_Anl = Ser.Is_Anl,
    @Cd_Ser = Ser.Cd_Ser
  FROM T_Tur Tur
    INNER JOIN T_Ser Ser ON Tur.Is_Ser = Ser.Is_Ser
  WHERE Tur.Is_Tur = @Is_Tur;

  SET @SerIni = CASE WHEN @Is_Anl < 20 THEN 15 ELSE 16 END;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ Limpa temporários residuais ]------------------------------- */
  DELETE T_Meb1
  WHERE IS_MAT BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND NR_MEB_BIM = @Nr_Bim;

  DELETE T_Mep1
  WHERE IS_MAT BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND NR_MEP_BIM = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ Recuperação Semestral / Semana Cultural ]-------------------- */
  EXEC Sp_Grv_Rec_Sem         @Is_Mat_Ini;
  EXEC Sp_Grv_SemanaCultural  @Is_Mat_Ini, @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ Apaga registros permanentes por turma/bimestre ]------------ */
  DELETE MEB
  FROM T_Meb MEB
    INNER JOIN T_Mat M ON M.Is_Mat = MEB.Is_Mat
  WHERE M.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND MEB.Nr_Meb_Bim = @Nr_Bim
    AND M.Is_Tur = @Is_Tur;

  DELETE MEP
  FROM T_Mep MEP
    INNER JOIN T_Mat M ON M.Is_Mat = MEP.Is_Mat
  WHERE M.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND MEP.Nr_Mep_Bim = @Nr_Bim
    AND M.Is_Tur = @Is_Tur;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 0 - Grv. Linha Média por Disciplina/Aluno (Cd_Ser @SerIni-23) ] */
  INSERT INTO T_Mep1
    (Is_Mat, Is_Dis, Is_Tmp, NR_MEP_PSO, Is_Dis_Dom,
     NR_MEP_BIM, TP_MEP_PRO, NR_MEP_QRT, Is_Usu, Dt_Sop)
  SELECT DISTINCT
    M.Is_Mat, Dse.Is_Dis, Mpg.Is_Tmp, Pmp.Nr_Pmp_Pso, Dse.Is_Dis_Dom,
    @Nr_Bim, 0,
    (SELECT Nr_Trt_Qtd FROM T_Trt
     WHERE Is_Tur = T.Is_Tur AND Is_Dis = Dse.Is_Dis
       AND Is_Tmp = Tmp.Is_Tmp AND Nr_Trt_Bim = @Nr_Bim),
    @Is_Usu, GETDATE()
  FROM T_Mat M
    INNER JOIN T_Tur T   ON M.Is_Tur = T.Is_Tur
    INNER JOIN T_Dse Dse ON T.Is_Ser = Dse.Is_Ser AND T.Tp_Tur_Esp = Dse.Tp_Dse_Tes
    INNER JOIN T_Ser Ser ON T.Is_Ser = Ser.Is_Ser
    INNER JOIN T_Ags Ags ON T.Is_Ser = Ags.Is_Ser
    INNER JOIN T_Tgs Tgs ON Ags.Is_Tgs = Tgs.Is_Tgs
    INNER JOIN T_Mag Mag ON Tgs.Is_Tgs = Mag.Is_Tgs
    INNER JOIN T_Mbg Mbg ON Mag.Is_Mag = Mbg.Is_Mag
    INNER JOIN T_Tmb Tmb ON Mbg.Is_Tmb = Tmb.Is_Tmb AND Tmb.Is_Anl = Ser.Is_Anl
    INNER JOIN T_Mpg Mpg ON Mbg.Is_Mbg = Mpg.Is_Mbg
    INNER JOIN T_Tmp Tmp ON Mpg.Is_Tmp = Tmp.Is_Tmp
    INNER JOIN T_Pmp Pmp ON Mpg.Is_Mpg = Pmp.Is_Mpg
  WHERE M.Is_Tur       = @Is_Tur
    AND Pmp.Nr_Pmp_Bim = @Nr_Bim
    AND M.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Ser.Cd_Ser BETWEEN @SerIni AND 23;

  /* ---[ 0 - Grv. Linha Média por Disciplina/Aluno (Cd_Ser 0-15) ]--- */
  INSERT INTO T_Mep1
    (Is_Mat, Is_Dis, Is_Tmp, NR_MEP_PSO, Is_Dis_Dom,
     NR_MEP_BIM, TP_MEP_PRO, NR_MEP_QRT, Is_Usu, Dt_Sop)
  SELECT DISTINCT
    M.Is_Mat, Dse.Is_Dis, Mpg.Is_Tmp, Pmp.Nr_Pmp_Pso, Dse.Is_Dis_Dom,
    @Nr_Bim, 0, Trt.Nr_Trt_Qtd, @Is_Usu, GETDATE()
  FROM T_Mat M
    INNER JOIN T_Tur T   ON M.Is_Tur = T.Is_Tur
    INNER JOIN T_Dse Dse ON T.Is_Ser = Dse.Is_Ser AND T.Tp_Tur_Esp = Dse.Tp_Dse_Tes
    INNER JOIN T_Ser Ser ON T.Is_Ser = Ser.Is_Ser
    INNER JOIN T_Ags Ags ON T.Is_Ser = Ags.Is_Ser
    INNER JOIN T_Tgs Tgs ON Ags.Is_Tgs = Tgs.Is_Tgs
    INNER JOIN T_Mag Mag ON Tgs.Is_Tgs = Mag.Is_Tgs
    INNER JOIN T_Mbg Mbg ON Mag.Is_Mag = Mbg.Is_Mag
    INNER JOIN T_Mpg Mpg ON Mbg.Is_Mbg = Mpg.Is_Mbg
    INNER JOIN T_Pmp Pmp ON Mpg.Is_Mpg = Pmp.Is_Mpg
    INNER JOIN T_Tmp Tmp ON Mpg.Is_Tmp = Tmp.Is_Tmp
    INNER JOIN T_Trt Trt ON Trt.Is_Tur  = T.Is_Tur
                         AND Trt.Is_Dis  = Dse.Is_Dis
                         AND Trt.Is_Tmp  = Tmp.Is_Tmp
                         AND Trt.Nr_Trt_Bim = @Nr_Bim
  WHERE M.Is_Tur       = @Is_Tur
    AND Pmp.Nr_Pmp_Bim = @Nr_Bim
    AND M.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Ser.Cd_Ser BETWEEN 0 AND 15;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ IS_NOT_OBR ]------------------------------------------------ */
  UPDATE T_Mep1
  SET IS_NOT_OBR = N.Is_Not
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON Mep.Is_Mat = M.Is_Mat
    INNER JOIN T_Tmp Tmp ON Tmp.Is_Tmp  = Mep.Is_Tmp
    INNER JOIN T_Mpg Mpg ON Tmp.Is_Tmp  = Mpg.Is_Tmp
    INNER JOIN T_Ntg Ntg ON Mpg.Is_Mpg  = Ntg.Is_Mpg
    INNER JOIN T_Not N   ON Mep.Is_Mat  = N.Is_Mat
                         AND Mep.Is_Dis  = N.Is_Dis
                         AND N.Nr_Not_Bim = Mep.Nr_Mep_Bim
                         AND N.Is_Tnt    = Ntg.Is_Tnt
  WHERE M.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND M.Is_Tur = @Is_Tur;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 1 - Qtd. realizadas e Soma por componente de média ]-------- */
  /* Otimização: QRA e Sma calculados em uma única passada na tabela   */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 1
     ,Nr_Mep_QRA = (SELECT DISTINCT COUNT(*)
        FROM T_Not N
          INNER JOIN T_Tnt Tnt ON N.Is_Tnt    = Tnt.Is_Tnt
          INNER JOIN T_Ntg Ntg ON Tnt.Is_Tnt  = Ntg.Is_Tnt
          INNER JOIN T_Mpg Mpg ON Ntg.Is_Mpg  = Mpg.Is_Mpg
          INNER JOIN T_Mbg Mbg ON Mpg.Is_Mbg  = Mbg.Is_Mbg
          INNER JOIN T_Mag Mag ON Mbg.Is_Mag  = Mag.Is_Mag
        WHERE N.Is_Mat      = M.Is_Mat
          AND N.Nr_Not_Bim  = @Nr_Bim
          AND N.Is_Dis      = Mep.Is_Dis
          AND Mpg.Is_Tmp    = Mep.Is_Tmp
          AND Mag.Is_Tgs    = Ags.Is_Tgs
          AND SUBSTRING(Cd_Tnt, 1, 1) <> 'P'   /* Retira Pontos */
        GROUP BY N.Is_Mat, N.Is_Dis, Ntg.Is_Mpg)
     ,Nr_Mep_Sma = (SELECT DISTINCT SUM(N.Nr_Not)
        FROM T_Not N
          INNER JOIN T_Tnt Tnt ON N.Is_Tnt    = Tnt.Is_Tnt
          INNER JOIN T_Ntg Ntg ON Tnt.Is_Tnt  = Ntg.Is_Tnt
          INNER JOIN T_Mpg Mpg ON Ntg.Is_Mpg  = Mpg.Is_Mpg
          INNER JOIN T_Mbg Mbg ON Mpg.Is_Mbg  = Mbg.Is_Mbg
          INNER JOIN T_Mag Mag ON Mbg.Is_Mag  = Mag.Is_Mag
        WHERE N.Is_Mat      = M.Is_Mat
          AND N.Nr_Not_Bim  = @Nr_Bim
          AND N.Is_Dis      = Mep.Is_Dis
          AND Mpg.Is_Tmp    = Mep.Is_Tmp
          AND Mag.Is_Tgs    = Ags.Is_Tgs)
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat    = Mep.Is_Mat
    INNER JOIN T_Tur Tur ON M.Is_Tur    = Tur.Is_Tur
    INNER JOIN T_Ags Ags ON Tur.Is_Ser  = Ags.Is_Ser
  WHERE M.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND M.Is_Tur        = @Is_Tur
    AND Mep.Nr_Mep_Bim  = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 2 - Média Digitada ]---------------------------------------- */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 2, NR_MEP_DIG = N.Nr_Not
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat       = Mep.Is_Mat
    INNER JOIN T_Tmp Tmp ON Mep.Is_Tmp     = Tmp.Is_Tmp
    INNER JOIN T_Not N   ON Tmp.Is_Tnt_Dig = N.Is_Tnt
                         AND N.Is_Dis      = Mep.Is_Dis
                         AND N.Is_Mat      = Mep.Is_Mat
                         AND N.Nr_Not_Bim  = @Nr_Bim
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 3 - Nota de Recuperação ]----------------------------------- */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 3, NR_MEP_REC = N.Nr_Not
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat       = Mep.Is_Mat
    INNER JOIN T_Tmp Tmp ON Mep.Is_Tmp     = Tmp.Is_Tmp
    INNER JOIN T_Not N   ON Tmp.Is_Tnt_Rec = N.Is_Tnt
                         AND N.Is_Dis      = Mep.Is_Dis
                         AND N.Is_Mat      = Mep.Is_Mat
                         AND N.Nr_Not_Bim  = @Nr_Bim
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 4 - Ponto na Média ]---------------------------------------- */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 4, NR_MEP_PTO = N.Nr_Not
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat       = Mep.Is_Mat
    INNER JOIN T_Tmp Tmp ON Mep.Is_Tmp     = Tmp.Is_Tmp
    INNER JOIN T_Not N   ON Tmp.Is_Tnt_Pto = N.Is_Tnt
                         AND N.Is_Dis      = Mep.Is_Dis
                         AND N.Is_Mat      = Mep.Is_Mat
                         AND N.Nr_Not_Bim  = Mep.Nr_Mep_Bim
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim = @Nr_Bim;

  UPDATE T_Mep1 SET Nr_Mep_Pto = 0 WHERE Nr_Mep_Pto IS NULL;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 4.1 - Ponto Simulado (fusão dos dois blocos IF em 1 UPDATE)] */
  /* @Cd_Ser <= 18 → Nr_Not >= 50; @Cd_Ser > 18 → Nr_Not = 50        */
  UPDATE T_Mep1
  SET Tp_Mep_Pro = 41
     ,Nr_Mep_Pto = Nr_Mep_Pto + 10
  FROM T_Mep1 Mep
    JOIN T_Not Nta ON Nta.Is_Mat      = Mep.Is_Mat
                   AND Mep.Is_Dis     = Nta.Is_Dis
                   AND Nta.Nr_Not_Bim = Mep.Nr_Mep_Bim
                   AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
                   AND Mep.Nr_Mep_Bim = @Nr_Bim
                   AND Mep.NR_MEP_PTO <= 90
    JOIN T_Tnt Tnt ON Tnt.Is_Tnt = Nta.Is_Tnt AND Tnt.Cd_Tnt = 'S1'
    JOIN T_Tmp Tmp ON Tmp.Is_Tmp  = Mep.Is_Tmp AND SUBSTRING(Cd_Tmp, 1, 1) = 'B'
  WHERE (@Cd_Ser <= 18 AND Nta.Nr_Not >= 50)
     OR (@Cd_Ser >  18 AND Nta.Nr_Not =  50);

  /* ---[ 4.2 - Ponto Simulado acima de 50 (somente Ensino Médio) ]--- */
  IF @Cd_Ser > 18
  BEGIN
    UPDATE T_Mep1
    SET Tp_Mep_Pro = 42
       ,Nr_Mep_Pto = Nr_Mep_Pto + 20
    FROM T_Mep1 Mep
      JOIN T_Not Nta ON Nta.Is_Mat      = Mep.Is_Mat
                     AND Mep.Is_Dis     = Nta.Is_Dis
                     AND Nta.Nr_Not_Bim = Mep.Nr_Mep_Bim
                     AND Nta.Nr_Not     > 50
                     AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
                     AND Mep.Nr_Mep_Bim = @Nr_Bim
                     AND Mep.NR_MEP_PTO <= 80
      JOIN T_Tnt Tnt ON Tnt.Is_Tnt = Nta.Is_Tnt AND Tnt.Cd_Tnt = 'S1'
      JOIN T_Tmp Tmp ON Tmp.Is_Tmp  = Mep.Is_Tmp AND SUBSTRING(Cd_Tmp, 1, 1) = 'B';
  END
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 5 - Cálculo da Média Parcial (3 UPDATEs → 1 com CASE) ]----- */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 5
     ,Nr_Mep_Cal = CASE
                     WHEN Mep.Nr_Mep_Pto IS NOT NULL
                          AND (Mep.Nr_Mep_Sma / Mep.Nr_Mep_Qra) + Mep.Nr_Mep_Pto > 100
                       THEN 100
                     WHEN Mep.Nr_Mep_Pto IS NOT NULL
                       THEN (Mep.Nr_Mep_Sma / Mep.Nr_Mep_Qra) + Mep.Nr_Mep_Pto
                     ELSE Mep.Nr_Mep_Sma / Mep.Nr_Mep_Qra
                   END
     ,Nr_Mep_Fin = CASE
                     WHEN Mep.Nr_Mep_Pto IS NOT NULL
                          AND (Mep.Nr_Mep_Sma / Mep.Nr_Mep_Qra) + Mep.Nr_Mep_Pto > 100
                       THEN 100
                     WHEN Mep.Nr_Mep_Pto IS NOT NULL
                       THEN (Mep.Nr_Mep_Sma / Mep.Nr_Mep_Qra) + Mep.Nr_Mep_Pto
                     ELSE Mep.Nr_Mep_Sma / Mep.Nr_Mep_Qra
                   END
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M ON M.Is_Mat = Mep.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.Nr_Mep_Pso  > 0                       /* Peso > ZERO        */
    AND Mep.Nr_Mep_Qra >= Mep.Nr_Mep_Qrt;         /* Qtd realiz >= turma*/
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 6 - Média Digitada final ]---------------------------------- */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 6, NR_MEP_FIN = Mep.Nr_Mep_Dig
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M ON M.Is_Mat = Mep.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.Nr_Mep_Dig IS NOT NULL;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 7 - Média com Recuperação ]--------------------------------- */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 7, NR_MEP_Fin = Mep.Nr_Mep_Rec
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat   = Mep.Is_Mat
    INNER JOIN T_Tmp Tmp ON Mep.Is_Tmp = Tmp.Is_Tmp
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.Nr_Mep_Rec IS NOT NULL
    AND Tmp.TP_TMP_REC  = '1'
    AND Mep.Nr_Mep_Rec  > Mep.NR_MEP_Fin;

  UPDATE T_Mep1
  SET TP_MEP_PRO = 7, NR_MEP_Fin = Tmb.NR_TMB_APR
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat   = Mep.Is_Mat
    INNER JOIN T_Tmp Tmp ON Mep.Is_Tmp = Tmp.Is_Tmp
    INNER JOIN T_Tsi Tsi ON Tmp.Is_Tsi = Tsi.Is_Tsi
    INNER JOIN T_Tmb Tmb ON Tsi.Is_Tsi = Tmb.Is_Tsi
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.Nr_Mep_Rec IS NOT NULL
    AND Tmp.TP_TMP_REC  = '2'
    AND Mep.Nr_Mep_Rec  > Mep.NR_MEP_Fin;

  UPDATE T_Mep1
  SET TP_MEP_PRO = 7, TP_MEP_REC = 'R '
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat   = Mep.Is_Mat
    INNER JOIN T_Tmp Tmp ON Mep.Is_Tmp = Tmp.Is_Tmp
    INNER JOIN T_Tsi Tsi ON Tmp.Is_Tsi = Tsi.Is_Tsi
    INNER JOIN T_Tmb Tmb ON Tsi.Is_Tsi = Tmb.Is_Tsi
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.Nr_Mep_Rec IS NOT NULL
    AND Mep.Nr_Mep_Rec >= Mep.NR_MEP_Fin
    AND Mep.Nr_Mep_Rec >= Tmb.NR_TMB_APR;

  UPDATE T_Mep1
  SET TP_MEP_PRO = 7, TP_MEP_REC = 'NR'
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat   = Mep.Is_Mat
    INNER JOIN T_Tmp Tmp ON Mep.Is_Tmp = Tmp.Is_Tmp
    INNER JOIN T_Tsi Tsi ON Tmp.Is_Tsi = Tsi.Is_Tsi
    INNER JOIN T_Tmb Tmb ON Tsi.Is_Tsi = Tmb.Is_Tsi
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.Nr_Mep_Rec IS NOT NULL
    AND ((Mep.Nr_Mep_Rec <  Mep.NR_MEP_Fin AND Mep.Nr_Mep_Fin  < Tmb.NR_TMB_APR)
      OR (Mep.Nr_Mep_Rec >= Mep.NR_MEP_Fin AND Mep.Nr_Mep_Rec  < Tmb.Nr_Tmb_Apr));
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 8 - Simbologia ]-------------------------------------------- */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 8, Is_Sim = Sim.Is_Sim
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat   = Mep.Is_Mat
    INNER JOIN T_Tur Tur ON M.Is_Tur   = Tur.Is_Tur
    INNER JOIN T_Ser Ser ON Tur.Is_Ser = Ser.Is_Ser
    INNER JOIN T_Sim Sim ON Sim.Is_Sim = (SELECT MAX(S.Is_Sim)
      FROM T_Tmp Tmp
        INNER JOIN T_Tsi Tsi ON Tmp.Is_Tsi = Tsi.Is_Tsi
        INNER JOIN T_Sim S   ON Tsi.Is_Tsi = S.Is_Tsi
      WHERE Mep.Is_Tmp = Tmp.Is_Tmp
        AND S.NR_SIM_NOT <= Mep.Nr_Mep_Cal
        AND Tsi.Tp_Tsi = 'S')
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.NR_Mep_Fin IS NOT NULL
    AND Ser.Cd_Ser BETWEEN 0 AND 15;

  UPDATE T_Mep1
  SET TP_MEP_PRO = 8, Is_Sim = Sim.Is_Sim
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M   ON M.Is_Mat   = Mep.Is_Mat
    INNER JOIN T_Tur Tur ON M.Is_Tur   = Tur.Is_Tur
    INNER JOIN T_Ser Ser ON Tur.Is_Ser = Ser.Is_Ser
    INNER JOIN T_Sim Sim ON Sim.Is_Sim = (SELECT MAX(S.Is_Sim)
      FROM T_Tmp Tmp
        INNER JOIN T_Tsi Tsi ON Tmp.Is_Tsi = Tsi.Is_Tsi
        INNER JOIN T_Sim S   ON Tsi.Is_Tsi = S.Is_Tsi
      WHERE Mep.Is_Tmp = Tmp.Is_Tmp
        AND S.NR_SIM_NOT <= Mep.Nr_Mep_Fin
        AND Tsi.Tp_Tsi = 'S')
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.NR_Mep_Fin IS NOT NULL
    AND Ser.Cd_Ser BETWEEN @SerIni AND 23;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 09 - Média Parcial — Disciplina Dominante ]----------------- */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 9
     ,Nr_Mep_Sma = (SELECT SUM(Nr_Mep_Fin)  FROM T_Mep1 Mep2
                    WHERE Mep2.Is_Dis_Dom = Mep.Is_Dis AND Mep2.Is_Mat = Mep.Is_Mat
                      AND Mep2.Is_Dis <> Mep2.Is_Dis_Dom AND Mep2.Is_Tmp = Mep.Is_Tmp)
     ,Nr_Mep_Qrt = (SELECT COUNT(*)          FROM T_Mep1 Mep2
                    WHERE Mep2.Is_Dis_Dom = Mep.Is_Dis AND Mep2.Is_Mat = Mep.Is_Mat
                      AND Mep2.Is_Dis <> Mep2.Is_Dis_Dom AND Mep2.Is_Tmp = Mep.Is_Tmp)
     ,Nr_Mep_Qra = (SELECT COUNT(*)          FROM T_Mep1 Mep2
                    WHERE Mep2.Is_Dis_Dom = Mep.Is_Dis AND Mep2.Is_Mat = Mep.Is_Mat
                      AND Mep2.Is_Dis <> Mep2.Is_Dis_Dom AND Mep2.Is_Tmp = Mep.Is_Tmp
                      AND Nr_Mep_Fin IS NOT NULL)
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M ON M.Is_Mat = Mep.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim = @Nr_Bim
    AND Mep.NR_Mep_Fin IS NULL
    AND Mep.Is_Sim     IS NULL
    AND Mep.Is_Dis      = Mep.Is_Dis_Dom
    AND EXISTS (SELECT Is_Dis_Dom FROM T_Mep1 Mep3
                WHERE Mep3.Is_Dis_Dom = Mep.Is_Dis_Dom AND Mep.Is_Dis <> Mep3.Is_Dis);
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 10 - Calcula Média — Disciplina Dominante ]----------------- */
  UPDATE T_Mep1
  SET TP_MEP_PRO = 32
     ,Nr_Mep_Fin = Nr_Mep_Sma / Nr_Mep_Qrt
     ,Nr_Mep_Cal = Nr_Mep_Sma / Nr_Mep_Qra
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M ON M.Is_Mat = Mep.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.NR_Mep_Fin IS NULL
    AND Mep.Is_Sim     IS NULL
    AND Mep.Is_Dis      = Mep.Is_Dis_Dom
    AND Mep.Nr_Mep_Qrt <= Mep.Nr_Mep_Qra;

  UPDATE T_Mep1
  SET TP_MEP_PRO = 33, Nr_Mep_Cal = 100, NR_MEP_FIN = 100
  FROM T_Mep1 Mep
    INNER JOIN T_Mat M ON M.Is_Mat = Mep.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Mep.Nr_Mep_Bim  = @Nr_Bim
    AND Mep.Nr_Mep_Pso  > 0
    AND Mep.Nr_Mep_Qra >= Mep.Nr_Mep_Qrt
    AND Mep.Nr_Mep_Pto IS NOT NULL
    AND (Mep.Nr_Mep_Sma / Mep.Nr_Mep_Qra) + Mep.Nr_Mep_Pto > 100;

  /* ---[ Cap global em 100 (fusão de 2 UPDATEs → 1) ]---------------- */
  UPDATE T_Mep1
  SET Nr_Mep_Cal = CASE WHEN Nr_Mep_Cal > 100 THEN 100 ELSE Nr_Mep_Cal END
     ,Nr_Mep_Fin = CASE WHEN Nr_Mep_Fin > 100 THEN 100 ELSE Nr_Mep_Fin END
  WHERE Nr_Mep_Cal > 100 OR Nr_Mep_Fin > 100;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ Grava T_Mep e limpa temporário ]---------------------------- */
  INSERT INTO T_Mep SELECT * FROM T_Mep1;

  DELETE T_Mep1
  FROM T_Mep1 Mep
    JOIN T_Mat Mat ON Mep.IS_MAT  = Mat.IS_MAT
    JOIN T_TUR     ON Mat.IS_TUR  = T_TUR.IS_TUR
  WHERE Mat.IS_TUR     = @Is_Tur
    AND MEP.NR_MEP_BIM = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 20 - Grv. Linha MEB por Disciplina/Aluno ]------------------ */
  INSERT INTO T_Meb1
    (Is_Dis, Is_Mat, Is_Tmb, Is_Mag, Nr_Meb_Pso,
     Tp_Meb_Pro, Nr_Meb_Bim, Is_Usu, Dt_Sop)
  SELECT DISTINCT
    Mep.Is_Dis, M.Is_Mat, Tmb.Is_Tmb, Mag.Is_Mag,
    (SELECT Nr_Pmb_Pso FROM T_Pmb Pmb
     WHERE Pmb.Is_Mbg = Mbg.Is_Mbg AND Pmb.Nr_Pmb_Bim = @Nr_Bim),
    20, @Nr_Bim, @Is_Usu, GETDATE()
  FROM T_Mep Mep
    INNER JOIN T_Mat M   ON Mep.Is_Mat  = M.Is_Mat
    INNER JOIN T_Tur T   ON M.Is_Tur    = T.Is_Tur
    INNER JOIN T_Ags Ags ON T.Is_Ser    = Ags.Is_Ser
    INNER JOIN T_Tgs Tgs ON Ags.Is_Tgs  = Tgs.Is_Tgs
    INNER JOIN T_Mag Mag ON Tgs.Is_Tgs  = Mag.Is_Tgs
    INNER JOIN T_Mbg Mbg ON Mag.Is_Mag  = Mbg.Is_Mag
    INNER JOIN T_Tmb Tmb ON Mbg.Is_Tmb  = Tmb.Is_Tmb
    INNER JOIN T_Pmb Pmb ON Pmb.IS_MBG  = Mbg.IS_MBG
  WHERE Mep.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND M.Is_Tur       = @Is_Tur
    AND Mep.Nr_Mep_Bim = @Nr_Bim
    AND Tmb.Is_Anl     = @Is_Anl
  GROUP BY M.Is_Mat, Mep.Is_Dis, Mep.Nr_Mep_Bim,
           Tmb.Is_Tmb, Mbg.Is_Mbg, Mag.Is_Mag;

  UPDATE T_Meb1
  SET Is_Dis_Dom = Mep.Is_Dis_Dom
  FROM T_Meb1 Meb
    INNER JOIN T_Mep Mep ON Meb.Is_Dis = Mep.Is_Dis AND Meb.Is_Mat = Mep.Is_Mat
  WHERE Mep.Nr_Mep_Bim = @Nr_Bim
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 21 - Qtd. Médias Necessárias / Realizadas ]----------------- */
  UPDATE T_Meb1
  SET TP_MEB_PRO = 21
     ,Nr_Meb_QPR = (SELECT COUNT(*) FROM T_Mep Mep
                    WHERE Mep.Is_Mat = Meb.Is_Mat AND Mep.Is_Dis = Meb.Is_Dis
                      AND Mep.Nr_Mep_Bim = Meb.Nr_Meb_Bim AND Mep.Nr_Mep_Fin IS NOT NULL)
     ,Nr_Meb_QPN = (SELECT COUNT(*) FROM T_Mep Mep
                    WHERE M.Is_Mat = Mep.Is_Mat AND Mep.NR_MEP_QRT > 0
                      AND Meb.Nr_Meb_Bim = Mep.Nr_Mep_Bim AND Meb.Is_Dis = Mep.Is_Dis)
  FROM T_Meb1 Meb
    INNER JOIN T_MAT M ON M.IS_MAT = Meb.IS_MAT
  WHERE Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND M.Is_Tur       = @Is_Tur
    AND Meb.Nr_Meb_Bim = @Nr_Bim
    AND NOT EXISTS (SELECT Is_Dis_Dom FROM T_MEB Mb2
                    WHERE Mb2.IS_DIS <> Mb2.IS_DIS_DOM
                      AND Mb2.IS_MAT       = Meb.IS_MAT
                      AND Mb2.NR_MEB_BIM   = Meb.NR_MEB_BIM);
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 23 - Soma Médias Parciais Realizadas ]----------------------- */
  /* Otimização: joins em vírgula das subqueries corrigidos p/ INNER JOIN */
  UPDATE T_Meb1
  SET TP_MEB_PRO = 23
     ,Nr_Meb_SMA = (SELECT SUM(Mep.Nr_Mep_Fin * Mep.NR_MEP_PSO)
        FROM T_Mep Mep
          INNER JOIN T_Tmp Tmp ON Mep.Is_Tmp = Tmp.Is_Tmp
        WHERE Mep.Is_Mat     = Meb.Is_Mat
          AND Tmp.Tp_Tmp_Med = 0
          AND Mep.Is_Dis     = Meb.Is_Dis
          AND Mep.Nr_Mep_Bim = Meb.Nr_Meb_Bim
          AND Mep.Nr_Mep_Fin IS NOT NULL
          AND Meb.Nr_Meb_Qpr >= Meb.Nr_Meb_Qpn)
     ,NR_MEB_SPP = (SELECT SUM(Mep.NR_MEP_PSO)
        FROM T_Mep Mep
          INNER JOIN T_Tmp Tmp ON Mep.Is_Tmp = Tmp.Is_Tmp
        WHERE Mep.Is_Mat     = Meb.Is_Mat
          AND Tmp.Tp_Tmp_Med = 0
          AND Mep.Is_Dis     = Meb.Is_Dis
          AND Mep.Nr_Mep_Bim = Meb.Nr_Meb_Bim
          AND Mep.Nr_Mep_Fin IS NOT NULL
          AND Meb.Nr_Meb_Qpr >= Meb.Nr_Meb_Qpn)
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M ON Meb.Is_Mat = M.Is_Mat
  WHERE Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND M.Is_Tur       = @Is_Tur
    AND Meb.Nr_Meb_Bim = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 24 - Média Digitada Bimestral ]----------------------------- */
  UPDATE T_Meb1
  SET TP_MEB_PRO = 24, NR_MEB_DIG = N.Nr_Not, NR_MEB_CAL = N.Nr_Not, NR_MEB_FIN = N.Nr_Not
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M   ON M.Is_Mat       = Meb.Is_Mat
    INNER JOIN T_Tmb Tmb ON Meb.Is_Tmb     = Tmb.Is_Tmb
    INNER JOIN T_Not N   ON Tmb.Is_Tnt_Dig = N.Is_Tnt
                         AND N.Is_Dis      = Meb.Is_Dis
                         AND N.Is_Mat      = Meb.Is_Mat
                         AND N.Nr_Not_Bim  = @Nr_Bim
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 25 - Ponto na Média Bimestral ]----------------------------- */
  UPDATE T_Meb1
  SET TP_MEB_PRO = 25, NR_MEB_PTO = N.Nr_Not
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M   ON M.Is_Mat       = Meb.Is_Mat
    INNER JOIN T_Tmb Tmb ON Meb.Is_Tmb     = Tmb.Is_Tmb
    INNER JOIN T_Not N   ON Tmb.Is_Tnt_Pto = N.Is_Tnt
                         AND N.Is_Dis      = Meb.Is_Dis
                         AND N.Is_Mat      = Meb.Is_Mat
                         AND N.Nr_Not_Bim  = @Nr_Bim
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 26 - Recuperação Bimestral ]-------------------------------- */
  UPDATE T_Meb1
  SET TP_MEB_PRO = 26, NR_MEB_REC = N.Nr_Not
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M   ON M.Is_Mat       = Meb.Is_Mat
    INNER JOIN T_Tmb Tmb ON Meb.Is_Tmb     = Tmb.Is_Tmb
    INNER JOIN T_Not N   ON Tmb.Is_Tnt_Rec = N.Is_Tnt
                         AND N.Is_Dis      = Meb.Is_Dis
                         AND N.Is_Mat      = Meb.Is_Mat
                         AND N.Nr_Not_Bim  = @Nr_Bim
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 27 - Cálculo da Média Bimestral (base) ]-------------------- */
  UPDATE T_Meb1
  SET TP_MEB_PRO = 27
     ,Nr_Meb_Cal = Meb.Nr_Meb_Sma / Meb.Nr_Meb_Spp
     ,Nr_Meb_Fin = Meb.Nr_Meb_Sma / Meb.Nr_Meb_Spp
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M ON M.Is_Mat = Meb.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim  = @Nr_Bim
    AND NR_MEB_FIN      IS NULL
    AND NR_MEB_CAL      IS NULL
    AND Meb.Nr_Meb_Spp  > 0                        /* Peso > ZERO        */
    AND Meb.NR_MEB_QPR >= Meb.NR_MEB_QPN           /* Qtd realiz >= turma*/
    AND (NOT EXISTS (SELECT Is_Dis_Dom FROM T_Meb1 Meb2
                     WHERE Meb2.Is_Dis <> Meb2.Is_Dis_Dom AND Meb2.Is_Mat = M.Is_Mat)
      OR EXISTS  (SELECT Is_Dis FROM T_Meb1 Meb3
                  WHERE Meb3.Is_Dis <> Meb.Is_Dis_Dom
                    AND Meb3.Is_Dis <> Meb3.Is_Dis_Dom AND Meb3.Is_Mat = M.Is_Mat));

  /* ---[ 27+Pto - Média com Ponto (IF/ELSE duplicado → 1 bloco) ]---- */
  /* Condição equivalente: executa exceto quando Cd_Ser=23 E Bim=2    */
  IF @Cd_Ser <> 23 OR @Nr_Bim <> 2
  BEGIN
    UPDATE T_Meb1
    SET TP_MEB_PRO = 5
       ,Nr_Meb_Cal = CASE
                       WHEN (Meb.Nr_Meb_Sma / Meb.Nr_Meb_Spp) + Meb.Nr_Meb_Pto > 100
                         THEN 100
                       ELSE (Meb.Nr_Meb_Sma / Meb.Nr_Meb_Spp) + Meb.Nr_Meb_Pto
                     END
       ,Nr_Meb_Fin = CASE
                       WHEN (Meb.Nr_Meb_Sma / Meb.Nr_Meb_Spp) + Meb.Nr_Meb_Pto > 100
                         THEN 100
                       ELSE (Meb.Nr_Meb_Sma / Meb.Nr_Meb_Spp) + Meb.Nr_Meb_Pto
                     END
    FROM T_Meb1 Meb
      INNER JOIN T_Mat M ON M.Is_Mat = Meb.Is_Mat
    WHERE M.Is_Tur = @Is_Tur
      AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
      AND Meb.Nr_Meb_Bim  = @Nr_Bim
      AND Meb.Nr_Meb_Spp  > 0
      AND Meb.NR_MEB_QPR >= Meb.NR_MEB_QPN
      AND Meb.Nr_Meb_Pto IS NOT NULL
      AND (NOT EXISTS (SELECT Is_Dis_Dom FROM T_Meb1 Meb2
                       WHERE Meb2.Is_Dis <> Meb2.Is_Dis_Dom AND Meb2.Is_Mat = M.Is_Mat)
        OR EXISTS  (SELECT Is_Dis FROM T_Meb1 Meb3
                    WHERE Meb3.Is_Dis <> Meb.Is_Dis_Dom
                      AND Meb3.Is_Dis <> Meb3.Is_Dis_Dom AND Meb3.Is_Mat = M.Is_Mat));
  END
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 29 - Faltas Digitadas ]------------------------------------- */
  UPDATE T_Meb1
  SET Nr_Meb_Flt = Nr_Faltas
  FROM Tb_Flt_Bim Flt
    JOIN T_Mat  Mat ON Mat.Is_Alu  = Flt.Is_Alu AND Mat.Is_Tur = Flt.Is_Tur
    JOIN T_Meb1 Meb ON Meb.Is_Mat  = Mat.Is_Mat AND Nr_Meb_Bim = Nr_Bim
    JOIN T_Dse  Dse ON Dse.Is_Dse  = Flt.Is_Dse AND Meb.Is_Dis = Dse.Is_Dis
    JOIN T_Dis  Dis ON Dis.Is_Dis  = Dse.Is_Dis
  WHERE Nr_Bim = @Nr_Bim;

  UPDATE T_Meb1
  SET TP_MEB_PRO = 29, NR_MEB_FLT = Fdd.Nr_Fdd
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M   ON M.Is_Mat       = Meb.Is_Mat
    INNER JOIN T_Fdd Fdd ON Meb.Is_Mat     = Fdd.Is_Mat
                         AND Meb.Is_Dis    = Fdd.Is_Dis
                         AND Meb.Nr_Meb_Bim = Fdd.Nr_Fdd_Bim
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 30 - Simbologia Bimestral ]--------------------------------- */
  UPDATE T_Meb1
  SET TP_MEb_PRO = 30, Is_Sim = Sim.Is_Sim
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M   ON M.Is_Mat   = Meb.Is_Mat
    INNER JOIN T_Sim Sim ON Sim.Is_Sim = (SELECT MAX(S.Is_Sim)
      FROM T_Tmb Tmb
        INNER JOIN T_Tsi Tsi ON Tmb.Is_Tsi = Tsi.Is_Tsi
        INNER JOIN T_Sim S   ON Tsi.Is_Tsi = S.Is_Tsi
      WHERE Meb.Is_Tmb = Tmb.Is_Tmb
        AND S.NR_SIM_NOT <= Meb.Nr_Meb_Fin
        AND Tsi.Tp_Tsi = 'S')
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim  = @Nr_Bim
    AND Meb.NR_Meb_Fin IS NOT NULL;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 31 - Média Bimestral — Disciplina Dominante ]--------------- */
  UPDATE T_Meb1
  SET TP_MEb_PRO = 31
     ,Nr_Meb_Sma = (SELECT SUM(Nr_Meb_Fin) FROM T_Meb1 Meb2
                    WHERE Meb2.Is_Dis_Dom = Meb.Is_Dis AND Meb2.Is_Mat = Meb.Is_Mat
                      AND Meb2.Is_Dis <> Meb2.Is_Dis_Dom)
     ,Nr_Meb_Flt = (SELECT SUM(Nr_Meb_Flt) FROM T_Meb1 Meb2
                    WHERE Meb2.Is_Dis_Dom = Meb.Is_Dis AND Meb2.Is_Mat = Meb.Is_Mat
                      AND Meb2.Is_Dis <> Meb2.Is_Dis_Dom AND Nr_Meb_Flt IS NOT NULL)
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M ON M.Is_Mat = Meb.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim  = @Nr_Bim
    AND Meb.Is_Sim      IS NULL
    AND Meb.Is_Dis       = Meb.Is_Dis_Dom
    AND Meb.Nr_Meb_Fin  IS NULL;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 32 - Qtd. Médias Bim. — Disciplina Dominante ]-------------- */
  UPDATE T_Meb1
  SET TP_MEb_PRO = 32
     ,Nr_Meb_Qpn = (SELECT COUNT(*) FROM T_Meb1 Meb2
                    WHERE Meb2.Is_Dis_Dom = Meb.Is_Dis AND Meb2.Is_Mat = Meb.Is_Mat
                      AND Meb2.Is_Dis <> Meb2.Is_Dis_Dom)
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M ON M.Is_Mat = Meb.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim = @Nr_Bim
    AND Meb.Is_Sim     IS NULL
    AND Meb.Is_Dis      = Meb.Is_Dis_Dom
    AND Meb.Nr_Meb_Fin IS NULL
    AND EXISTS (SELECT Is_Dis_Dom FROM T_MEB1 Mb2
                WHERE Mb2.IS_DIS <> Mb2.IS_DIS_DOM
                  AND Mb2.IS_MAT       = Meb.IS_MAT
                  AND Mb2.NR_MEB_BIM   = Meb.NR_MEB_BIM);
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 33 - Qtd. Realizadas — Disciplina Dominante ]--------------- */
  UPDATE T_Meb1
  SET TP_MEb_PRO = 33
     ,Nr_Meb_Qpr = (SELECT COUNT(*) FROM T_Meb1 Meb2
                    WHERE Meb2.Is_Dis_Dom = Meb.Is_Dis AND Meb2.Is_Mat = Meb.Is_Mat
                      AND Meb2.Is_Dis <> Meb2.Is_Dis_Dom AND Nr_Meb_Fin IS NOT NULL)
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M ON M.Is_Mat = Meb.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim = @Nr_Bim
    AND Meb.Is_Sim     IS NULL
    AND Meb.Is_Dis      = Meb.Is_Dis_Dom
    AND Meb.Nr_Meb_Fin IS NULL
    AND EXISTS (SELECT Is_Dis_Dom FROM T_Meb1
                WHERE Is_Dis <> Is_Dis_Dom AND Is_Dis_Dom = Meb.Is_Dis);
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ 35 - Calcula Média — Disciplina Dominante ]----------------- */
  UPDATE T_Meb1
  SET TP_MEb_PRO = 35
     ,Nr_Meb_Fin = Nr_Meb_Sma / Nr_Meb_Qpr
     ,Nr_Meb_Cal = Nr_Meb_Sma / Nr_Meb_Qpr
  FROM T_Meb1 Meb
    INNER JOIN T_Mat M ON M.Is_Mat = Meb.Is_Mat
  WHERE M.Is_Tur = @Is_Tur
    AND Meb.Is_Mat BETWEEN @Is_Mat_Ini AND @Is_Mat_Fin
    AND Meb.Nr_Meb_Bim  = @Nr_Bim
    AND Meb.NR_Meb_Fin IS NULL
    AND Meb.Is_Sim     IS NULL
    AND Meb.Is_Dis      = Meb.Is_Dis_Dom
    AND Meb.Nr_Meb_Qpn  = Meb.Nr_Meb_Qpr;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  /* ---[ Grava T_Meb e limpa temporário ]---------------------------- */
  INSERT INTO T_Meb SELECT * FROM T_Meb1;

  DELETE T_Meb1
  FROM T_Meb1 Meb
    JOIN T_Mat Mat ON Meb.IS_MAT = Mat.IS_MAT
    JOIN T_TUR     ON Mat.IS_TUR = T_TUR.IS_TUR
  WHERE Mat.IS_TUR     = @Is_Tur
    AND MEB.NR_MEB_BIM = @Nr_Bim;
  /* ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^ */

  COMMIT TRANSACTION;
END TRY
BEGIN CATCH
  IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
  DECLARE @ErrMsg  NVARCHAR(4000) = ERROR_MESSAGE();
  DECLARE @ErrSev  INT            = ERROR_SEVERITY();
  RAISERROR(@ErrMsg, @ErrSev, 1);
END CATCH
GO
